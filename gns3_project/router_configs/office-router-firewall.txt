service firewall stop
service dnsmasq stop
service network stop

uci set system.@system[0].hostname='office-router-firewall'
uci commit

service system restart

uci set network.wan=interface
uci set network.wan.device='eth1'
uci set network.wan.proto='static'
uci set network.wan.ipaddr='192.168.122.2'
uci set network.wan.netmask='255.255.255.0'
uci commit

uci set network.vlan10=interface
uci set network.vlan10.device='eth2.10'
uci set network.vlan10.proto='static'
uci set network.vlan10.ipaddr='192.168.10.1'
uci set network.vlan10.netmask='255.255.255.0'
uci set network.vlan10.ip6assign='60'
uci commit

service network restart

uci set dhcp.vlan10=dhcp
uci set dhcp.vlan10.interface='vlan10'
uci set dhcp.vlan10.start='2'
uci set dhcp.vlan10.limit='253'
uci set dhcp.vlan10.leasetime='12h'
uci set dhcp.vlan10.dhcpv4='server'
uci set dhcp.vlan10.dhcpv6='server'
uci set dhcp.vlan10.ra='server'
uci set dhcp.vlan10.ra_slaac='1'
uci add_list dhcp.vlan10.ra_flags='managed-config'
uci add_list dhcp.vlan10.ra_flags='other-config'
uci add_list dhcp.vlan10.dhcp_option='6,192.168.10.1'
uci commit

service dnsmasq restart

uci add firewall zone
uci set firewall.@zone[-1].name='vlan10'
uci set firewall.@zone[-1].network='vlan10'
uci set firewall.@zone[-1].input='ACCEPT'
uci set firewall.@zone[-1].output='ACCEPT'
uci set firewall.@zone[-1].forward='ACCEPT'
uci commit

uci add firewall forwarding
uci set firewall.@forwarding[-1]=forwarding
uci set firewall.@forwarding[-1].src='vlan10'
uci set firewall.@forwarding[-1].dest='wan'
uci commit

service firewall restart
